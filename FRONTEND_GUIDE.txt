━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  まず「何を触らなければいいか」を理解する
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  このアプリの「触っていい部分」と「触らない部分」を整理する。

  ┌────────────────────────────────┐
  │  触っていいファイル（フロントエンド）  │
  │                                  │
  │  public/index.html               │  ← 画面の構造・骨格
  │  public/style.css                │  ← 色・フォント・レイアウト
  │  public/app.js ※一部注意あり     │  ← ボタンの動き・表示ロジック
  └────────────────────────────────┘

  ┌────────────────────────────────┐
  │  触ってはいけないファイル          │
  │  （バックエンド＝AIとの通信部分）  │
  │                                  │
  │  server.js                       │
  │  lib/analyzer.js                 │
  │  lib/preprocessor.js             │
  │  data/palmistry-db.js            │
  └────────────────────────────────┘

  ルール: public/ フォルダの中だけ変えれば、見た目は自由に変えられる。


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  app.js について「触っていい部分」「注意が必要な部分」
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  app.js は「動きをつくるファイル」なので、一部はバックエンドと連動している。

  【自由に変えていい】
  ・ボタンを押したときのアニメーション
  ・エラーメッセージの文言
  ・結果カードの表示フォーマット（どこに何を表示するか）
  ・ローディング中のテキスト・ステップ表示
  ・履歴の表示デザイン

  【変えると壊れる可能性あり（注意）】
  ・fetch('/analyze', ...) ← サーバーへの通信部分。URLや送るデータを変えるな
  ・pollResult()           ← ポーリングの仕組み。触るな
  ・renderResult(data)     ← dataの中身の構造はAIが決めている。フィールド名を変えるな
  ・FormData に append している項目名 ← サーバー側と一致している必要がある

  要するに: 「APIとの通信」「データの受け取り」は触らない。
           「受け取ったデータをどう見せるか」は自由。


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  手順：ローカルで作業する環境を作る
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Step 1. Node.js をインストールする
    https://nodejs.org から LTS版をダウンロードしてインストール。
    ターミナル（Macならターミナル.app、WindowsならPowerShell）で確認:
      node -v    → バージョンが表示されれば成功
      npm -v     → npmもセットで入る

  Step 2. リポジトリをダウンロードする
    ターミナルで:
      git clone https://github.com/ryufekyo1204-hub/tesou.git
      cd tesou

  Step 3. 必要な道具をインストールする
      npm install
    （package.json に書いてある全ライブラリが node_modules/ に入る）

  Step 4. APIキーを設定する
    プロジェクトのフォルダに .env というファイルを作る:
      ANTHROPIC_API_KEY=sk-ant-xxxxxxxxxxxxxxxx
    ※ .env は Git に含まれない（.gitignoreに設定済み）。
       絶対にGitHubにアップロードしないこと。

  Step 5. ローカルで起動する
      npm start   または   node server.js
    ターミナルに「起動中 → http://localhost:3000」と出たら成功。
    ブラウザで http://localhost:3000 を開く。

  Step 6. ファイルを変えて → ブラウザをリロードして確認
    style.css を変えたら: ブラウザで Ctrl+R (Mac: Cmd+R) でリロード
    server.js/analyzer.js を変えたら: ターミナルで Ctrl+C → 再度 node server.js


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  手順：実際に変更を加えてみる（練習）
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ── 練習1: テーマカラーを変える ──

  style.css の先頭にある :root{ } ブロックを探す。
  CSS変数という仕組みで、アプリ全体の色が管理されている。

    :root {
      --purple:    #7c3aed;   ← メインカラー（紫）
      --purple-lt: #a78bfa;   ← 明るい紫
      --gold:      #fbbf24;   ← 金色
      --green:     #4ade80;   ← 緑
      --bg:        #0c0820;   ← 背景色（ほぼ黒）
      --text:      #f1f0ff;   ← テキスト色（白っぽい）
    }

  --purple を #1d4ed8（青）に変えると → ボタンや枠が全部青になる。
  --bg を #0a1628（濃紺）に変えると → 背景が変わる。

  ポイント: #RRGGBB の形式。オンラインで「color picker」と検索すると
  好みの色のコードを調べられる。

  ── 練習2: フォントを変える ──

  Googleフォント（https://fonts.google.com）で気に入ったフォントを選ぶ。
  例: 「Noto Sans JP」を使う場合

  index.html の <head> に追加:
    <link href="https://fonts.googleapis.com/css2?family=Noto+Sans+JP&display=swap" rel="stylesheet">

  style.css の body に追加:
    body { font-family: 'Noto Sans JP', sans-serif; }

  ── 練習3: ボタンの文言を変える ──

  index.html で「🔮 手相を鑑定する」を探す。
  そのまま好きな文字に変えるだけ。

  ── 練習4: カードの背景色を変える ──

  style.css で .card を探す。
  background や border-color を変える。


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  HTML・CSS・JS の役割分担（超基礎）
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  家で例えると:
  HTML = 間取り図（どこに何の部屋があるか）
  CSS  = インテリア・内装（色・形・配置）
  JS   = 家の動き（自動ドアが開く、照明が点く）

  ── HTMLの基本構造 ──

    <div class="card">        ← div は「箱」。class で名前をつける
      <h3>タイトル</h3>        ← 見出し
      <p>本文テキスト</p>       ← 段落
      <button id="btn">送信</button>  ← ボタン。id で固有名
    </div>

  ── CSSの書き方 ──

    .card {                   ← クラス名（.card）に対してスタイルを定義
      background: #1a1030;    ← 背景色
      border-radius: 12px;    ← 角丸
      padding: 16px;          ← 内側の余白
      color: white;           ← 文字色
    }

    #btn {                    ← IDに対するスタイル（. ではなく # を使う）
      font-size: 16px;
    }

  ── JSの書き方（このアプリで使っているパターン） ──

    // IDで要素を取得する
    const btn = document.getElementById('btn');

    // クリックされたときの処理
    btn.addEventListener('click', () => {
      // ここに処理を書く
      alert('クリックされました');
    });

    // テキストを変える
    document.getElementById('openingText').textContent = '新しいテキスト';

    // クラスを追加・削除して表示/非表示を切り替える
    element.classList.add('hidden');     // 非表示
    element.classList.remove('hidden'); // 表示

  デプロイ（公開）の手順
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ローカルで変更を確認したら、Render に反映させる。

  Step 1. 変更をGitHubに送る
    ターミナルで:
      git add public/style.css        ← 変更したファイルを選択
      git add public/index.html       ← 複数ある場合はまとめて
      git commit -m "デザインを変更"   ← 変更内容のメモ
      git push origin main            ← GitHubへアップロード

  Step 2. Renderが自動でデプロイする
    GitHubのmainブランチに変更が入ると、Renderが自動で検知して
    新しいバージョンを公開してくれる（2〜5分かかる）。

  Step 3. 確認
    Renderのダッシュボードでデプロイ完了を確認。
    本番URLをブラウザで開いて動作確認。

  ポイント: 「git add → git commit → git push」の3ステップが基本動作。
  これをGitのコミットフローと言う。

  よくある作業パターン集
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ── パターンA: 「アプリ名・キャッチコピーを変えたい」 ──

  触るファイル: index.html のみ

  検索: 「手相鑑定」「AI による本格手相分析」で検索して書き換える。
  ヘッダーロゴの絵文字も変えられる（🔮 → 🌙 など）。

  ── パターンB: 「配色をガラッと変えたい（ダークではなくライトテーマ）」 ──

  触るファイル: style.css のみ

  :root の変数を変える:
    --bg:     #ffffff   （背景を白に）
    --text:   #1a1a1a   （文字を黒に）
    --surface: #f5f5f5  （カード背景を薄いグレーに）
    --purple:  #6d28d9  （アクセントカラーはそのままでも可）

  ── パターンC: 「結果の表示順序を変えたい」 ──

  触るファイル: index.html のみ

  #resultSection 内のカードの並び順を変えるだけ。
  HTMLの要素を上下に移動させる = 表示順が変わる。

  例: 「ラッキー」を「人生アドバイス」の前に出したい
    → index.html で .card-lucky の div ブロックを
       .card-advice の div ブロックより上に移動する。

  ── パターンD: 「新しいセクション・カードを追加したい」 ──

  触るファイル: index.html + style.css + app.js

  1. index.html に新しいカードのHTMLを追加（既存カードをコピーして改変）
  2. style.css に新しいカードのスタイルを追加（必要なら）
  3. app.js の renderResult() 関数内で、新しいカードにデータを流し込む
     例: setText('myNewCardId', data.interpretation.some_field);

  ── パターンE: 「スマホとPCで別のレイアウトにしたい」 ──

  触るファイル: style.css のみ

  メディアクエリという仕組みを使う:
    @media (min-width: 768px) {
      /* 画面幅768px以上（PC・タブレット）の時だけ適用されるCSS */
      .card {
        max-width: 600px;
        margin: 0 auto;
      }
    }

  このアプリは現在スマホ優先（モバイルファースト）のデザイン。

  ブラウザの開発者ツールを使う（最重要スキル）
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  フロントエンド開発で最も重要なツール。無料で今すぐ使える。

  開き方:
    Mac:     Cmd + Option + I
    Windows: F12 または Ctrl + Shift + I
    または: 画面を右クリック → 「検証」または「要素を調査」

  使い方:

  【Elementsタブ（HTMLを確認・変更）】
    画面の任意の場所を右クリック → 「検証」
    → その要素のHTMLがハイライトされる
    → ダブルクリックで直接テキストを変更できる（一時的な変更）
    → CSSも右側のパネルで即座に変更・確認できる

  【Consoleタブ（エラーを確認）】
    JavaScriptのエラーがここに表示される。
    赤い文字 = エラー（要対応）
    黄色い文字 = 警告（要注意）
    白い文字 = ログ（情報）

    このアプリで何か動かなくなったら、まずConsoleを見る。

  【Networkタブ（APIの通信を確認）】
    サーバーとの通信が全部見える。
    /analyze にリクエストが飛んでいるか、
    /result/:jobId のレスポンスが正しいか確認できる。

  ポイント: 開発者ツールでの変更はリロードすると消える。
           確認できたら、ファイルに反映させる。

  やらかしやすいミス集（注意事項）
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  【HTMLの </div> を消した】
  開いた <div> を閉じる </div> が対応していないと、
  レイアウトが崩れる。ブラウザは「なんとかしよう」とするが
  予測できない見た目になる。
  → 開発者ツールの Elements で確認。VS Code の括弧ハイライトを活用。

  【CSSのセレクタ名を間違えた】
  .card と .Card は別物（大文字・小文字が区別される）。
  変更が効かない場合はタイポを疑う。
  → 開発者ツールの Styles パネルで打ち消し線が出ていたら無効になっている。

  【JSでエラーが出て画面が動かなくなった】
  JavaScriptはエラーが出るとそれ以降の処理が全部止まる。
  → ブラウザのConsoleを開いてエラーを確認する。

  【変えたのに反映されない】
  ブラウザがキャッシュ（古いファイル）を使っていることがある。
  → 強制リロード: Ctrl+Shift+R（Mac: Cmd+Shift+R）

  【app.js の fetch URL を変えてしまった】
  fetch('/analyze', ...) の '/analyze' を変えると通信できなくなる。
  server.js と一致している必要がある。

  【.env ファイルをGitHubにアップした】
  APIキーが流出する。すぐAnthropicのダッシュボードでキーを無効化する。
  .gitignore に .env が書かれていれば自動で除外されるが要確認。
